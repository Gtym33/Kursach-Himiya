	id = rcw_fareasresp.1
	title = rcw_fareasresp.1.t
	desc = rcw_fareasresp.1.desc
	fire_only_once = yes
	picture = GFX_report_event_RSS_siberian_clique
	trigger = {
        RUS = { 
		   has_country_flag = SOV_czechs_evac_way_north
				
			NOT = { has_country_flag = SOV_czechs_evac_first_echelon_completed }
			NOT = { has_country_flag = SOV_czechs_evac_echelons_loading_completed }
			NOT = { has_country_flag = SOV_czechs_evac_last_echelon_completed }
        }
		OR = {
			AND = {
				country_exists = BKL
		        AND = {
		            tag = BKL
				    OR = { has_war_with = RUS has_war_with = FER }
		        }
			}
			AND = {
				NOT = { country_exists = BKL }
		        AND = {
		            tag = HRB
					HRB = { is_subject = no }
		        }
			}
		}
	}
	
	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = rcw_fareasresp.1.a
		custom_effect_tooltip = far_eastern_volunteer_tt
        hidden_effect = {
            division_template = {
				name = "Dalnevostochnye dobrovolcy"
				regiments = {
					militia = { x = 0 y = 0 }
					militia = { x = 0 y = 1 }
				}
				support = {
				}
			}
            division_template = {
				name = "Dalnevostochnye kazaki"
				regiments = {
					irregular_cavalry = { x = 0 y = 0 }
					irregular_cavalry = { x = 0 y = 1 }
				}
				support = {
				}
			}
			capital_scope = {
				create_unit = {
					division = "name = \"Dalnevostochnye dobrovolcy\" division_template = \"Dalnevostochnye dobrovolcy\" start_experience_factor = 0.08 start_equipment_factor = 1.0"
					owner = PREV
                    count = 2
				} 
                create_unit = {
					division = "name = \"Dalnevostochnye kazaki\" division_template = \"Dalnevostochnye kazaki\" start_experience_factor = 0.2 start_equipment_factor = 1.0"
					owner = PREV
                    count = 2
				}

        }
	}
}

#Постановление о прекращении фронтовой борьбы 
country_event = {
	id = rcw_fareasresp.2
	title = rcw_fareasresp.2.t
	desc = rcw_fareasresp.2.desc
	fire_only_once = yes
	picture = GFX_report_event_RSS_siberian_clique
	trigger = {
        tag = FER
		AND = {
			OR = {
				has_government = communism
				has_government = radical_socialism
			}
			OR = { 
				is_subject_of = RUS
				is_in_faction_with = RUS
			}
			OR = {
				NOT = { country_exists = AMR }
                NOT = {
					AMR = { 
						OR = {
				            has_government = communism
					        has_government = radical_socialism
				        }
				    } 
				}
			}
			FER = { NOT = { owns_state = 408 owns_state = 409 } }
		}
	}
	
	mean_time_to_happen = {
		days = 10
	}
	
	option = {
		name = rcw_fareasresp.2.a
		add_war_support = -1
		add_stability = -1
		add_ideas = disarmed_nation
		delete_unit = {
            disband = yes
        }
		rise_collapse_of_railway_system_3 = yes
		rise_collapse_of_railway_system = yes
		retire_character = RUS_Konstantin_Sukhanov
		if = {
			limit = { country_exists = AMR }
			AMR = { add_ideas = FER_red_partisans }
		}
	}
	option = {
		name = rcw_fareasresp.2.b
		add_political_power = -200
		add_stability = -0.3
		add_war_support = -0.3
		add_ideas = volunteer_only
		transfer_units_fraction = {
			target = REB
			size = 0.5
			stockpile_ratio = 0.5
			army_ratio = 0.5
			navy_ratio = 0.5
			air_ratio = 0.5
			keep_unit_leaders_trigger = {
			    always = yes
			}
		}
		rise_collapse_of_railway_system = yes
		rise_collapse_of_railway_system = yes
		retire_character = RUS_Konstantin_Sukhanov
	}
}
