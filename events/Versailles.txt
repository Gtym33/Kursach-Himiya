				if = {
					limit = {
						OR {
							check_variable = { global.petrograd_war_conference_vote_support_RUS > global.petrograd_war_conference_vote_not_support_RUS }
							AND = {
								NOT = { check_variable = { global.petrograd_war_conference_vote_support_RUS < global.petrograd_war_conference_vote_not_support_RUS } }

						RUS = { country_event = { id = petrograd_war_conference_Entente.5 days = 1 } } # Отказ в поддержке наступления

				}

			limit = { tag = FRA }
			country_event = petrograd_war_conference_Entente.8 #Секретный договор с Россией
		}
		add_to_variable = { global.petrograd_war_conference_vote_support_RUS = 1 }
		if = {
			limit = {
				any_other_country = { has_country_flag = petrograd_war_conference_member
